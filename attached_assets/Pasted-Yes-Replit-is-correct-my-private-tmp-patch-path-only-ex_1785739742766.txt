async function setLocale(context: BrowserContext, locale: Locale) {
  // Protected dashboard routes are not locale-prefixed route trees in this app.
  // src/middleware.ts strips locale prefixes for auth checks but does not
  // rewrite /es/referrer/dashboard to /referrer/dashboard. The layouts read
  // oku_locale, so the regression matrix pins locale through the cookie.
  await context.addCookies([
    {
      name: "oku_locale",
      value: locale,
      domain: "localhost",
      path: "/",
      httpOnly: false,
      secure: false,
      sameSite: "Lax",
    },
  ]);
}

async function stubSharedConsoleNetwork(page: Page) {

  await page.route("**/api/v1/me/beneficiary", async (route) => {

  });
}

async function expectNoRawKeysOrStrayLabels(page: Page) {
  const visibleText = await page.locator("body").innerText();
  expect(
    visibleText,
    "visible dashboard text must not contain raw i18n keys",
  ).not.toMatch(/console\.title|console\.tab\.|streetForm\.guestQR\./);

  const html = await page.content();
  expect(html, "old stray QR tooltip/title must stay removed").not.toContain("Booking QR Code");
}

async function expectInfluencerConsoleBeforePerformance(page: Page, consoleTitle: string) {
  const order = await page.evaluate((title) => {
    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_ELEMENT);
    let consoleIndex = -1;
    let performanceIndex = -1;
    let i = 0;

    while (walker.nextNode()) {
      const el = walker.currentNode as HTMLElement;
      const text = (el.textContent || "").trim();
      if (consoleIndex === -1 && text === title) consoleIndex = i;
      if (performanceIndex === -1 && text === "Performance") performanceIndex = i;
      if (consoleIndex !== -1 && performanceIndex !== -1) break;
      i += 1;
    }

    return { consoleIndex, performanceIndex };
  }, consoleTitle);

  expect(order.consoleIndex, "influencer console heading should render").toBeGreaterThanOrEqual(0);
  expect(order.performanceIndex, "influencer Performance section should render").toBeGreaterThanOrEqual(0);
  expect(order.consoleIndex, "QR console must appear before Performance in DOM order").toBeLessThan(
    order.performanceIndex,
  );
}

        await setLocale(context, locale);
        await loginAs(context, profile.email);
        await stubSharedConsoleNetwork(page);
        await profile.setup(page);

        await page.goto(profile.route, { waitUntil: "domcontentloaded" });

        const copy = COPY[locale];
        await expect(page.getByRole("heading", { name: copy.consoleTitle }).first()).toBeVisible();
        await expect(page.getByText(copy.qrTab).first()).toBeVisible();
        await expect(page.getByText(copy.instruction).first()).toBeVisible();
        await expect(page.getByText(copy.whatsapp).first()).toBeVisible();

        const qrCode = page.locator('svg[width="248"][height="248"]').first();
        await expect(qrCode).toBeVisible();
        await expectNoRawKeysOrStrayLabels(page);

        const noResults = page.getByText(/No results found/i);
        await expect(noResults, "whole-dashboard empty state must not replace QR console").toHaveCount(0);

        const stats = page.getByText(/Performance/i).first();
        if (await stats.isVisible().catch(() => false)) {
          await expectInfluencerConsoleBeforePerformance(page, copy.consoleTitle);
          const [qrBox, statsBox] = await Promise.all([
            qrCode.boundingBox(),
            stats.boundingBox(),
          ]);
          expect(qrBox, "QR code should have layout").not.toBeNull();
          expect(statsBox, "Performance stats should have layout").not.toBeNull();
          expect(
            statsBox!.y,
            `${profile.name} must not show private stats above the QR on ${testInfo.project.name}`,
          ).toBeGreaterThan(qrBox!.y);
        }
